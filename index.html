flowchart TD
    START([Customer arrives]) --> A[Enter app / visit store]

    %% AUTH
    A --> A1{Has account?}
    A1 -- No, skip --> B[Browse menu]
    A1 -- Yes, sign in --> A2[Authenticate SEIUE / email]
    A2 --> A3{Valid session?}
    A3 -- No --> A4[Login failed]
    A4 --> A5{Customer action}
    A5 -- Retry --> A2
    A5 -- Continue as guest --> B
    A3 -- Yes --> A6[Load profile + loyalty + credit]
    A6 --> A7{Credit overdue?}
    A7 -- Yes --> A8[Show locked account banner]
    A8 --> A9{Customer action}
    A9 -- Settle overdue tab --> A10[Process settlement]
    A10 --> A6
    A9 -- View only mode --> B
    A7 -- No --> B

    %% BROWSE
    B --> B1[Browse menu · filter / search / sort]
    B1 --> C{Item selected?}
    C -- No, continue --> B1
    C -- Yes --> C1{In stock?}
    C1 -- No --> C2[Show Sold out]
    C2 --> C3{Customer action}
    C3 -- Browse alternatives --> B1
    C3 -- Set back-in-stock alert --> C4([Notification])
    C3 -- Leave --> EXIT1([Exit])
    C1 -- Yes --> D[Add to cart + qty + customization]

    %% CART
    D --> E{More items?}
    E -- Yes --> B1
    E -- No --> F[Review cart]

    F --> F1{Cart actions}
    F1 -- Edit quantity --> F2[Update · clamp to stock]
    F2 --> F
    F1 -- Edit customization --> F3[Update preferences]
    F3 --> F
    F1 -- Remove item --> F
    F1 -- Clear cart --> F4{Confirm clear?}
    F4 -- Confirm --> F5[Cart emptied]
    F5 --> E
    F4 -- Cancel --> F
    F1 -- Apply loyalty --> F6[Discount applied]
    F6 --> F
    F1 -- Checkout --> G[Begin checkout]
    F1 -- Save cart --> F7[localStorage]

    %% CHECKOUT
    G --> G1{Identity}
    G1 -- New customer --> G2[Enter name / email / phone]
    G2 --> G3[Account created at checkout]
    G1 -- Existing --> G4[Select saved profile]
    G4 --> G3
    G3 --> H[Select pickup time]
    H --> H1{Order type}
    H1 -- Takeaway --> H2[Set collection time]
    H1 -- Dine-in --> H3[Assign table / token]
    H1 -- Delivery --> H4[Enter address]
    H4 --> H5{Address serviceable?}
    H5 -- No --> H6[Unavailable area]
    H6 --> H7{Customer action}
    H7 -- Switch to pickup --> H2
    H7 -- Edit address --> H4
    H7 -- Cancel --> CANCEL1([Abandoned])
    H5 -- Yes --> I[Choose payment]
    H2 --> I; H3 --> I

    %% PAYMENT
    I --> P1{Payment method}
    P1 -- Card / Wallet --> P2[Process online payment]
    P1 -- Pay at counter --> P3[Mark paid · no capture]
    P1 -- Credit tab (赊账) --> P4[Tab selected]

    P4 --> P5[Choose 1-14 days]
    P5 --> P6{Authenticated?}
    P6 -- No --> P7[Login required]
    P7 --> P8{Customer action}
    P8 -- Sign in --> A2
    P8 -- Other method --> I
    P6 -- Yes --> P9[Schedule due date]

    P1 -- Gift card --> GFT[Validate card]
    GFT --> GFT1{Valid?}
    GFT1 -- No --> GFT2[Invalid]
    GFT2 --> GFT3{Customer action}
    GFT3 -- Retry --> GFT
    GFT3 -- Other method --> I
    GFT1 -- Yes --> P2

    P2 --> P10{Gateway success?}
    P10 -- No --> P11[Show failure reason]
    P11 --> P12{Customer action}
    P12 -- Retry same --> P2
    P12 -- Different card --> I
    P12 -- Pay at counter --> P3
    P12 -- Cancel --> PAYCANCEL([Cancelled + not captured])
    P10 -- Yes --> J[Payment confirmed]
    P3 --> J; P9 --> J

    %% ORDER CREATION
    J --> K[Create order · validate stock + total]
    K --> K1{Stock sufficient at commit?}
    K1 -- No --> K2[Reject · stock conflict]
    K2 --> K3{Customer action}
    K3 -- Review cart --> F
    K3 -- Cancel --> CANCEL2([Cancelled])
    K1 -- Yes --> K4[Deduct inventory + persist]

    %% KITCHEN
    K4 --> L[Order sent to barista queue]
    L --> L1{Store accepts?}
    L1 -- No --> L2[Order rejected]
    L2 --> L3{Resolution}
    L3 -- Auto-refund --> L4[Refund initiated]
    L3 -- Offer substitute --> L5{Customer accepts?}
    L5 -- Yes --> L[Re-queue]
    L5 -- No --> L6[Cancel + refund]
    L4 --> L6
    L1 -- Yes --> M[Prepare drinks / food]
    M --> M1{Prep issue?}
    M1 -- Yes --> M2[Delay / partial outage]
    M2 --> M3{Customer decision}
    M3 -- Wait --> M
    M3 -- Accept replacement --> M
    M3 -- Cancel --> L6
    M1 -- No --> N[Quality check + pack]

    %% FULFILLMENT
    N --> O{Fulfillment mode}
    O -- Dine-in --> O1[Serve at table / call token]
    O -- Takeaway --> O2[Notify ready for pickup]
    O -- Delivery --> O3[Assign rider + dispatch]

    O3 --> O4{Delivered?}
    O4 -- Yes --> O5[Confirmed]
    O4 -- No --> O6[Re-contact / redelivery]
    O6 --> O7{Resolution}
    O7 -- Delivered --> O5
    O7 -- Failed --> O8[Compensate / refund]
    O1 --> X([Completed])
    O2 --> X; O5 --> X; O8 --> X

    %% POST-ORDER
    X --> POST{Post-order action}
    POST -- Rate / review --> POST1[Submit feedback]
    POST -- Report issue --> POST2[Support ticket]
    POST2 --> POST3{Resolution}
    POST3 -- Refund --> X
    POST3 -- Remake --> M
    POST -- Reorder --> B1
    POST -- Account activity --> ACT[Account page]
    ACT --> POST
    POST -- Nothing --> END([Exit])
    POST1 --> END

    %% CREDIT LIFECYCLE
    X -.-> CR{Was order a tab?}
    CR -- Yes --> CR1[Track due date]
    CR1 --> CR2{Due passed?}
    CR2 -- Yes --> CR3[Account locked → A7]
    CR2 -- No --> POST
